    function refreshCinematicLines() {

      isSynced = syncedLines.length > 0;
      cinematicLines = isSynced ? syncedLines.map(item => item.text) : getLines(lyricsInput.value);

    // ===== TAP SYNC TOOL =====
    // Lets the admin play the song and tap a button in rhythm to capture a
    // [mm:ss.xx] timestamp per lyric line, instead of typing times by hand.
    function stripTimeTags(line) {
      return line.replace(/\[(\d{1,2}):(\d{2}(?:\.\d{1,3})?)\]/g, "").trim();
    }

    function formatTapTag(seconds) {
      const minutes = Math.floor(seconds / 60);
      const rest = (seconds - minutes * 60).toFixed(2).padStart(5, "0");
      return `[${String(minutes).padStart(2, "0")}:${rest}]`;
    }

    function updateTapTimer() {
      if (!tapTimerEl) return;
      const minutes = Math.floor(livePositionSeconds / 60);
      const seconds = (livePositionSeconds % 60).toFixed(1).padStart(4, "0");
      tapTimerEl.textContent = `${String(minutes).padStart(2, "0")}:${seconds}`;
    }

    function renderTapState() {
      tapProgress.textContent = `${tapResults.length} / ${tapQueue.length} baris`;
      tapCurrentEl.textContent = tapCursor < tapQueue.length ? tapQueue[tapCursor] : "🎉 semua baris sudah ditandai";
      tapNextEl.textContent = tapCursor + 1 < tapQueue.length ? `berikutnya: ${tapQueue[tapCursor + 1]}` : "";
      tapButton.disabled = tapCursor >= tapQueue.length;
      tapUndoBtn.disabled = tapResults.length === 0;
      tapSkipBtn.disabled = tapCursor >= tapQueue.length;
      tapApplyBtn.disabled = tapResults.length === 0;
    }

    function openTapSync() {
      const rawLines = getLines(lyricsInput.value).map(stripTimeTags).filter(Boolean);
      if (!rawLines.length) {
        showToast("Tulis dulu baris liriknya di editor (boleh tanpa tag waktu)");
        lyricsInput.focus();
        return;
      }
      tapQueue = rawLines;
      tapResults = [];
      tapCursor = 0;
      tapTrackName.textContent = `${songs[selectedIndex].title} · ${songs[selectedIndex].artist}`;
      renderTapState();
      updateTapTimer();
      tapStage.classList.add("open");
      tapStage.setAttribute("aria-hidden", "false");
      document.body.style.overflow = "hidden";
      window.clearInterval(tapTimerInterval);
      tapTimerInterval = window.setInterval(updateTapTimer, 100);
    }

    function closeTapSync() {
      tapStage.classList.remove("open");
      tapStage.setAttribute("aria-hidden", "true");
      document.body.style.overflow = "";
      window.clearInterval(tapTimerInterval);
    }

    function tapNow() {
      if (tapCursor >= tapQueue.length) return;
      tapResults.push({ time: livePositionSeconds, text: tapQueue[tapCursor] });
      tapCursor += 1;
      tapButton.classList.remove("pulse");
      void tapButton.offsetWidth;
      tapButton.classList.add("pulse");
      renderTapState();
    }

    function undoTap() {
      if (!tapResults.length) return;
      tapResults.pop();
      tapCursor = Math.max(0, tapCursor - 1);
      renderTapState();
    }

    function skipTapLine() {
      if (tapCursor >= tapQueue.length) return;
      tapCursor += 1;
      renderTapState();
    }

    function applyTapResults() {
      if (!tapResults.length) return;
      lyricsInput.value = tapResults.map(item => `${formatTapTag(item.time)} ${item.text}`).join("\n");
      refreshCinematicLines();
      lyricStatus.textContent = lyricsInput.value.trim() ? "unsaved draft" : "not available";
      showToast(`${tapResults.length} baris berhasil di-sync — jangan lupa Save text`);
      closeTapSync();
    }

    function getTrackId(value) {
      const input = value.trim();
      const fromUrl = input.match(/spotify\.com\/track\/([A-Za-z0-9]{22})/i);
      if (fromUrl) return fromUrl[1];
      return /^[A-Za-z0-9]{22}$/.test(input) ? input : "";
    }

    function getSongColors(seed) {
      const palette = [
        ["#ff7a93", "#ffbd7e", "rgba(255, 122, 147, 0.21)"],
        ["#7cc7ff", "#a795ff", "rgba(124, 199, 255, 0.21)"],
        ["#65e0c1", "#8da7ff", "rgba(101, 224, 193, 0.21)"],
        ["#ffc96a", "#ff7e9b", "rgba(255, 201, 106, 0.21)"]
      ];
      const total = [...seed].reduce((sum, character) => sum + character.charCodeAt(0), 0);
      const [accent, accent2, soft] = palette[total % palette.length];
      return { accent, accent2, soft };
    }

    function addCustomSong(data, options = {}) {
      if (!data?.trackId || songs.some(song => song.trackId === data.trackId)) return;
      songs.push({ title: data.title, artist: data.artist, trackId: data.trackId, ...getSongColors(data.trackId), custom: true });
      renderShelf();
      if (options.select) selectSong(songs.length - 1);
    }

    async function loadCustomSongs() {
      Object.values(savedCustomSongs).forEach(song => addCustomSong(song));
      try {
        const snapshot = await getDocs(collection(db, "customSongs"));
        snapshot.forEach(item => {
          const song = item.data();
          if (typeof song.title !== "string" || typeof song.artist !== "string" || typeof song.trackId !== "string") return;
          savedCustomSongs[song.trackId] = song;
          addCustomSong(song);
        });
        localStorage.setItem(customSongKey, JSON.stringify(savedCustomSongs));
      } catch (error) {
        console.error("Could not load custom songs", error);
      }
    }
