    async function loadLyricsForSong(song) {
      const requestId = ++lyricRequest;
      lyricStatus.textContent = "loading cloud";
      try {
        const snapshot = await getDoc(doc(db, "lyrics", song.trackId));
        if (requestId !== lyricRequest) return;
        const cloudText = snapshot.exists() && typeof snapshot.data().text === "string"
          ? snapshot.data().text
          : "";
        const text = cloudText || savedLyrics[song.trackId] || "";
        lyricsInput.value = text;
        if (text) {
          savedLyrics[song.trackId] = text;
          localStorage.setItem(lyricKey, JSON.stringify(savedLyrics));
        }
        refreshCinematicLines();
        lyricStatus.textContent = cloudText ? "cloud saved" : "not available";
        syncState.classList.remove("error");
      } catch (error) {
        if (requestId !== lyricRequest) return;
        lyricsInput.value = savedLyrics[song.trackId] || "";
        refreshCinematicLines();
        lyricStatus.textContent = lyricsInput.value.trim() ? "local fallback" : "cloud unavailable";
        syncState.classList.add("error");
        console.error("Could not load lyrics from Firestore", error);
      }
    }

    async function toggleAuthentication() {
      try {
        if (auth.currentUser) {
          await signOut(auth);
          showToast("Signed out");
        } else {
          await signInWithPopup(auth, provider);
          showToast("Google sign-in berhasil");
        }
      } catch (error) {
        const message = error.code === "auth/unauthorized-domain"
          ? "Tambahkan domain Vercel ke Firebase Authorized domains"
          : "Login Google belum berhasil. Coba lagi.";
        showToast(message);
        console.error("Firebase authentication error", error);
      }
    }

    function triggerLightning() {
      if (!document.body.classList.contains("storm-room")) return;
      stormLayer.classList.remove("strike");
      stormFlash.classList.remove("flash");
      void stormLayer.offsetWidth;
      stormLayer.classList.add("strike");
      stormFlash.classList.add("flash");
      window.setTimeout(() => {
        stormLayer.classList.remove("strike");
        stormFlash.classList.remove("flash");
      }, 720);
    }

    function scheduleLightning() {
      window.clearTimeout(stormTimer);
      if (!document.body.classList.contains("storm-room")) return;
      stormTimer = window.setTimeout(() => {
        triggerLightning();
        scheduleLightning();
      }, 3600 + Math.random() * 5200);
    }

    function setRoomMood(song) {
      const isStormRoom = song.special === "storm";
      document.body.classList.toggle("storm-room", isStormRoom);
      focusEyebrow.textContent = isStormRoom ? "special room · storm mode" : "now selected";
      cinemaLabel.textContent = isStormRoom ? "special room · thunder cinematic" : "cinematic lyric mode";
      window.clearTimeout(stormTimer);
      if (isStormRoom) {
        triggerLightning();
        scheduleLightning();
      }
    }

    function renderShelf() {
      const query = search.value.trim().toLowerCase();
      const filtered = songs.map((song, index) => ({ song, index })).filter(({ song }) =>
        `${song.title} ${song.artist}`.toLowerCase().includes(query)
      );
      grid.textContent = "";
      filtered.forEach(({ song, index }) => {
        const card = document.createElement("button");
        card.type = "button";
        card.className = `song-card${index === selectedIndex ? " active" : ""}${song.special ? " special" : ""}`;
        card.style.setProperty("--card-accent", song.accent);
        card.innerHTML = `<small>${String(index + 1).padStart(2, "0")}</small><strong>${song.title}</strong><span>${song.artist}</span>`;
        card.addEventListener("click", () => selectSong(index));
        grid.appendChild(card);
      });
      emptyFilter.style.display = filtered.length ? "none" : "block";
      document.getElementById("song-count").textContent = query ? `${filtered.length} found` : `${songs.length} tracks`;
    }

    function renderBars() {
      const visualizer = document.getElementById("visualizer");
      visualizer.textContent = "";
      for (let i = 0; i < 38; i += 1) {
        const bar = document.createElement("i");
        bar.style.setProperty("--height", `${8 + Math.random() * 25}px`);
        bar.style.setProperty("--wave", `${0.55 + Math.random() * 0.75}s`);
        bar.style.setProperty("--delay", `${-Math.random() * 1.2}s`);
        visualizer.appendChild(bar);
      }
    }

    function renderPlayer(song) {
      if (spotifyController) {
        spotifyController.loadUri(`spotify:track:${song.trackId}`);
        return;
      }
      // API belum ready saat song dipilih (misal langsung ganti lagu pas load awal):
      // simpan sebagai pending, dan tampilkan embed biasa dulu biar tetap keliatan.
      pendingTrackId = song.trackId;
      spotifyPlayer.innerHTML = `<iframe title="${song.title} by ${song.artist} on Spotify" src="https://open.spotify.com/embed/track/${song.trackId}?utm_source=generator&theme=0" width="100%" height="152" loading="lazy" allow="autoplay; clipboard-write; encrypted-media; fullscreen; picture-in-picture"></iframe>`;
    }

    function selectSong(index) {
      selectedIndex = index;
      const song = songs[index];
      document.documentElement.style.setProperty("--accent", song.accent);
      document.documentElement.style.setProperty("--accent-2", song.accent2);
      document.documentElement.style.setProperty("--accent-soft", song.soft);
      setRoomMood(song);
      focusName.textContent = song.title;
      focusArtist.textContent = song.artist;
      coverLetter.textContent = song.title.charAt(0);
      lyricsInput.value = savedLyrics[song.trackId] || "";
      lyricStatus.textContent = lyricsInput.value.trim() ? "local cache" : "loading cloud";
      refreshCinematicLines();
      cinemaTrackName.textContent = `${song.title} · ${song.artist}`;
      renderPlayer(song);
      renderBars();
      renderShelf();
      loadLyricsForSong(song);
    }

    saveLyricsButton.addEventListener("click", async () => {
      if (!auth.currentUser) {
        await toggleAuthentication();
        return;
      }
      const song = songs[selectedIndex];
      const text = lyricsInput.value.trim();
      saveLyricsButton.disabled = true;
      lyricStatus.textContent = "saving cloud";
      try {
        await setDoc(doc(db, "lyrics", song.trackId), {
          text,
          updatedAt: serverTimestamp()
        }, { merge: true });
        if (text) savedLyrics[song.trackId] = text;
        else delete savedLyrics[song.trackId];
        localStorage.setItem(lyricKey, JSON.stringify(savedLyrics));
        lyricStatus.textContent = text ? "cloud saved" : "cleared";
        showToast(text ? "Lirik tersimpan ke cloud" : "Lirik cloud dikosongkan");
      } catch (error) {
        lyricStatus.textContent = "save denied";
        const message = error.code === "permission-denied"
          ? "Akun ini tidak punya izin untuk menyimpan"
          : "Cloud gagal menyimpan. Coba lagi.";
        showToast(message);
        console.error("Could not save lyrics to Firestore", error);
      } finally {
        saveLyricsButton.disabled = !activeUser;
      }
    });
