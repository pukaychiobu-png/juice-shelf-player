    function getLines(text) {

    // Parses LRC-style "[mm:ss.xx] line text" tags. A line can carry more than
    // one time tag (repeated chorus lines), same as the standard .lrc format.
    function parseSyncedLyrics(text) {
      const timeTagPattern = /\[(\d{1,2}):(\d{2}(?:\.\d{1,3})?)\]/g;
      const parsed = [];
      text.split(/\n+/).forEach(raw => {
        const line = raw.trim();
        if (!line) return;
        const times = [];
        let match;
        timeTagPattern.lastIndex = 0;
        while ((match = timeTagPattern.exec(line))) {
          times.push(parseInt(match[1], 10) * 60 + parseFloat(match[2]));
        }
        const cleanText = line.replace(timeTagPattern, "").trim();
        if (times.length && cleanText) times.forEach(time => parsed.push({ time, text: cleanText }));
      });
      parsed.sort((a, b) => a.time - b.time);
      return parsed;
    }

    function updateLiveLyricLine(index) {
      if (!liveLyric) return;
      if (!isSynced || index < 0 || !syncedLines[index]) {
        liveLyric.classList.remove("show");
        liveLyric.textContent = "";
        return;
      }
      liveLyric.textContent = syncedLines[index].text;
      liveLyric.classList.add("show");
    }

    function findSyncedIndex(positionSeconds) {
      if (!isSynced) return -1;
      let index = -1;
      for (let i = 0; i < syncedLines.length; i += 1) {
        if (syncedLines[i].time <= positionSeconds) index = i; else break;
      }
      return index;
    }

    // Called on every Spotify playback_update tick with the live position.
    function applySyncedLyricPosition(positionSeconds) {
      if (!isSynced) return;
      const index = findSyncedIndex(positionSeconds);
      if (index === -1 || index === lastSyncedIndex) return;
      lastSyncedIndex = index;
      cinematicIndex = index;
      updateLiveLyricLine(index);
      if (cinematicStage.classList.contains("open")) renderCinematic();
    }
