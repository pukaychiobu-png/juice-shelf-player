    addSongForm.addEventListener("submit", async event => {
      event.preventDefault();
      if (!auth.currentUser) {
        await toggleAuthentication();
        return;
      }
      const trackId = getTrackId(newSongUrl.value);
      const title = newSongTitle.value.trim();
      const artist = newSongArtist.value.trim();
      if (!trackId) {
        showToast("Tempel link Spotify track yang valid");
        newSongUrl.focus();
        return;
      }
      if (!title || !artist) {
        showToast("Isi judul dan artist dulu");
        return;
      }
      if (songs.some(song => song.trackId === trackId)) {
        showToast("Lagu itu sudah ada di shelf");
        return;
      }
      const customSong = { title, artist, trackId };
      try {
        await setDoc(doc(db, "customSongs", trackId), {
          ...customSong,
          updatedAt: serverTimestamp()
        }, { merge: true });
        savedCustomSongs[trackId] = customSong;
        localStorage.setItem(customSongKey, JSON.stringify(savedCustomSongs));
        addSongForm.reset();
        addSongForm.classList.remove("open");
        addCustomSong(customSong, { select: true });
        showToast("Lagu baru masuk ke shelf");
      } catch (error) {
        const message = error.code === "permission-denied"
          ? "Tambahkan aturan Firestore untuk customSongs"
          : "Gagal menyimpan lagu. Coba lagi.";
        showToast(message);
        console.error("Could not save custom song", error);
      }
    });

    function renderCinematic() {
      if (!cinematicLines.length) return;
      const line = cinematicLines[cinematicIndex];
      cinematicLine.textContent = "";
      line.split(/\s+/).forEach((word, index) => {
        const wordElement = document.createElement("span");
        wordElement.className = "cinematic-word";
        wordElement.style.setProperty("--word-index", index);
        wordElement.textContent = word;
        cinematicLine.appendChild(wordElement);
        if (index < line.trim().split(/\s+/).length - 1) cinematicLine.appendChild(document.createTextNode(" "));
      });
      lyricPrevious.textContent = cinematicLines[cinematicIndex - 1] || "";
      lyricNext.textContent = cinematicLines[cinematicIndex + 1] || "";
      cinemaProgress.textContent = `${String(cinematicIndex + 1).padStart(2, "0")} / ${String(cinematicLines.length).padStart(2, "0")}`;
      constellation.textContent = "";
      const dotCount = Math.min(cinematicLines.length, 12);
      for (let dotIndex = 0; dotIndex < dotCount; dotIndex += 1) {
        const dot = document.createElement("i");
        const activeIndex = cinematicLines.length <= 12
          ? cinematicIndex
          : Math.round((cinematicIndex / Math.max(cinematicLines.length - 1, 1)) * (dotCount - 1));
        if (dotIndex === activeIndex) dot.classList.add("active");
        constellation.appendChild(dot);
      }
      cinematicLine.classList.remove("reveal");
      lyricTapZone.classList.remove("reveal");
      void cinematicLine.offsetWidth;
      cinematicLine.classList.add("reveal");
      lyricTapZone.classList.add("reveal");
      if (cinemaHint) {
        cinemaHint.textContent = isSynced
          ? "mengikuti beat lagu secara otomatis · tap untuk override manual"
          : "tap layar atau tekan → untuk lanjut";
      }
    }

    function openCinematic() {
      refreshCinematicLines();
      if (!cinematicLines.length) {
        showToast("Tambahkan baris teks dulu untuk mode cinematic");
        lyricsInput.focus();
        return;
      }
      cinematicIndex = 0;
      renderCinematic();
      cinematicStage.classList.add("open");
      cinematicStage.setAttribute("aria-hidden", "false");
      document.body.style.overflow = "hidden";
      lyricTapZone.focus();
    }

    function closeCinematic() {
      cinematicStage.classList.remove("open");
      cinematicStage.setAttribute("aria-hidden", "true");
      document.body.style.overflow = "";
      document.getElementById("open-cinematic").focus();
    }

    function moveCinematic(direction) {
      if (!cinematicLines.length) return;
      const nextIndex = cinematicIndex + direction;
      if (nextIndex < 0 || nextIndex >= cinematicLines.length) {
        showToast(nextIndex < 0 ? "Ini baris pertama" : "Sudah sampai baris terakhir");
        return;
      }
      cinematicIndex = nextIndex;
      renderCinematic();
    }

    document.getElementById("open-cinematic").addEventListener("click", openCinematic);
    document.getElementById("close-cinematic").addEventListener("click", closeCinematic);
    document.getElementById("previous-line").addEventListener("click", () => moveCinematic(-1));
    document.getElementById("next-line").addEventListener("click", () => moveCinematic(1));
    lyricTapZone.addEventListener("click", () => moveCinematic(1));

      if (!cinematicStage.classList.contains("open")) return;

      if (event.key === "ArrowRight" || event.key === " ") { event.preventDefault(); moveCinematic(1); }
      if (event.key === "ArrowLeft") { event.preventDefault(); moveCinematic(-1); }
    });
