    // Parses LRC-style "[mm:ss.xx] line text" tags. A line can carry more than
    // one time tag (repeated chorus lines), same as the standard .lrc format.
    function parseSyncedLyrics(text) {
      const timeTagPattern = /\[(\d{1,2}):(\d{2}(?:\.\d{1,3})?)\]/g;
      const parsed = [];
      text.split(/\n+/).forEach(raw => {
        const line = raw.trim();
        if (!line) return;
        const times = [];
        let match;
        timeTagPattern.lastIndex = 0;
        while ((match = timeTagPattern.exec(line))) {
          times.push(parseInt(match[1], 10) * 60 + parseFloat(match[2]));
        }
        const cleanText = line.replace(timeTagPattern, "").trim();
        if (times.length && cleanText) times.forEach(time => parsed.push({ time, text: cleanText }));
      });
      parsed.sort((a, b) => a.time - b.time);
      return parsed;
    }

    function findSyncedIndex(positionSeconds) {
      if (!isSynced) return -1;
      let index = -1;
      for (let i = 0; i < syncedLines.length; i += 1) {
        if (syncedLines[i].time <= positionSeconds) index = i; else break;
      }
      return index;
    }

    // Called on every Spotify playback_update tick with the live position.
    function applySyncedLyricPosition(positionSeconds) {
      if (!isSynced) return;
      const index = findSyncedIndex(positionSeconds);
      if (index === -1 || index === lastSyncedIndex) return;
      lastSyncedIndex = index;
      cinematicIndex = index;
      updateLiveLyricLine(index);
      if (cinematicStage.classList.contains("open")) renderCinematic();
    }

    // Lets the admin play the song and tap a button in rhythm to capture a
    // [mm:ss.xx] timestamp per lyric line, instead of typing times by hand.
    function stripTimeTags(line) {
      return line.replace(/\[(\d{1,2}):(\d{2}(?:\.\d{1,3})?)\]/g, "").trim();
    }

    function formatTapTag(seconds) {
      const minutes = Math.floor(seconds / 60);
      const rest = (seconds - minutes * 60).toFixed(2).padStart(5, "0");
      return `[${String(minutes).padStart(2, "0")}:${rest}]`;
    }

    function getTrackId(value) {
      const input = value.trim();
      const fromUrl = input.match(/spotify\.com\/track\/([A-Za-z0-9]{22})/i);
      if (fromUrl) return fromUrl[1];

    }

    function getSongColors(seed) {
      const palette = [
        ["#ff7a93", "#ffbd7e", "rgba(255, 122, 147, 0.21)"],
        ["#7cc7ff", "#a795ff", "rgba(124, 199, 255, 0.21)"],
        ["#65e0c1", "#8da7ff", "rgba(101, 224, 193, 0.21)"],
        ["#ffc96a", "#ff7e9b", "rgba(255, 201, 106, 0.21)"]
      ];
      const total = [...seed].reduce((sum, character) => sum + character.charCodeAt(0), 0);
      const [accent, accent2, soft] = palette[total % palette.length];
      return { accent, accent2, soft };
    }

    }

    async function loadCustomSongs() {
      Object.values(savedCustomSongs).forEach(song => addCustomSong(song));
      try {
        const snapshot = await getDocs(collection(db, "customSongs"));

      }
    }
